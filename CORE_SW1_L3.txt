hostname CORE-SW1
no ip domain-lookup
ip domain-name qvis.local

ip routing                                 ! turn on L3 forwarding

username netadmin privilege 15 secret 9 $9$REPLACE-WITH-STRONG-SECRET
enable secret 9 $9$REPLACE-WITH-STRONG-SECRET
service password-encryption
!
banner motd ^ QVIS - Authorised Access Only - Activity Logged ^
!
crypto key generate rsa modulus 2048
ip ssh version 2
!
line con 0
 logging synchronous

line vty 0 15

! ============================================================
! EAST-WEST POLICY  (Named extended ACLs on SVIs, inbound)
! ============================================================
!
! ---- EXEC-FIN: most restricted user VLAN -----------------
ip access-list extended ACL-EXEC-FIN-IN
 remark allow DNS and DHCP to DCs
 permit udp 10.10.30.0 0.0.0.255 host 10.10.20.10 eq 53
 permit udp 10.10.30.0 0.0.0.255 host 10.10.20.11 eq 53
 permit udp 10.10.30.0 0.0.0.255 host 10.10.20.10 eq 67
 permit udp 10.10.30.0 0.0.0.255 host 10.10.20.11 eq 67
 remark allow AD authentication, LDAP, Kerberos, SMB to DCs
 permit tcp 10.10.30.0 0.0.0.255 10.10.20.0 0.0.0.255 eq 88
 permit udp 10.10.30.0 0.0.0.255 10.10.20.0 0.0.0.255 eq 88
 permit tcp 10.10.30.0 0.0.0.255 10.10.20.0 0.0.0.255 eq 389
 permit tcp 10.10.30.0 0.0.0.255 10.10.20.0 0.0.0.255 eq 636
 permit tcp 10.10.30.0 0.0.0.255 10.10.20.0 0.0.0.255 eq 445
 permit tcp 10.10.30.0 0.0.0.255 10.10.20.0 0.0.0.255 range 49152 65535
 remark allow print
 permit tcp 10.10.30.0 0.0.0.255 10.10.80.0 0.0.0.255 eq 9100
 permit tcp 10.10.30.0 0.0.0.255 10.10.80.0 0.0.0.255 eq 631
 remark allow voice signalling
 permit udp 10.10.30.0 0.0.0.255 10.10.70.0 0.0.0.255 range 16384 32767
 permit tcp 10.10.30.0 0.0.0.255 10.10.70.0 0.0.0.255 eq 5060
 remark deny everything else to internal RFC1918
 deny   ip 10.10.30.0 0.0.0.255 10.10.0.0 0.0.255.255
 remark allow Internet (gets translated by R1)
 permit ip 10.10.30.0 0.0.0.255 any
!
! ---- STAFF: same shape as EXEC-FIN, lower sensitivity ----
ip access-list extended ACL-STAFF-IN
 remark same allow list as EXEC-FIN
 permit udp 10.10.40.0 0.0.0.255 10.10.20.0 0.0.0.255 eq 53
 permit udp 10.10.40.0 0.0.0.255 10.10.20.0 0.0.0.255 eq 67
 permit tcp 10.10.40.0 0.0.0.255 10.10.20.0 0.0.0.255 eq 88
 permit udp 10.10.40.0 0.0.0.255 10.10.20.0 0.0.0.255 eq 88
 permit tcp 10.10.40.0 0.0.0.255 10.10.20.0 0.0.0.255 eq 389
 permit tcp 10.10.40.0 0.0.0.255 10.10.20.0 0.0.0.255 eq 636
 permit tcp 10.10.40.0 0.0.0.255 10.10.20.0 0.0.0.255 eq 445
 permit tcp 10.10.40.0 0.0.0.255 10.10.20.0 0.0.0.255 range 49152 65535
 permit tcp 10.10.40.0 0.0.0.255 10.10.80.0 0.0.0.255 eq 9100
 permit tcp 10.10.40.0 0.0.0.255 10.10.80.0 0.0.0.255 eq 631
 permit udp 10.10.40.0 0.0.0.255 10.10.70.0 0.0.0.255 range 16384 32767
 permit tcp 10.10.40.0 0.0.0.255 10.10.70.0 0.0.0.255 eq 5060
 remark explicitly deny STAFF reaching EXEC-FIN
 deny   ip 10.10.40.0 0.0.0.255 10.10.30.0 0.0.0.255
 deny   ip 10.10.40.0 0.0.0.255 10.10.50.0 0.0.0.255
 deny   ip 10.10.40.0 0.0.0.255 10.10.60.0 0.0.0.255
 permit ip 10.10.40.0 0.0.0.255 any
!
! ---- LAB: outbound to internet only, nothing else --------
ip access-list extended ACL-LAB-IN
 remark allow DHCP/DNS to DC1 only (no AD join)
 permit udp 10.10.60.0 0.0.0.255 host 10.10.20.10 eq 53
 permit udp 10.10.60.0 0.0.0.255 host 10.10.20.10 eq 67
 remark deny LAB -> any RFC1918
 deny   ip 10.10.60.0 0.0.0.255 10.0.0.0 0.255.255.255
 deny   ip 10.10.60.0 0.0.0.255 172.16.0.0 0.15.255.255
 deny   ip 10.10.60.0 0.0.0.255 192.168.0.0 0.0.255.255
 permit ip 10.10.60.0 0.0.0.255 any
!
! ---- GUEST: internet only ---------------------------------
ip access-list extended ACL-GUEST-IN
 remark deny GUEST -> any RFC1918
 deny   ip 10.10.90.0 0.0.0.255 10.0.0.0 0.255.255.255
 deny   ip 10.10.90.0 0.0.0.255 172.16.0.0 0.15.255.255
 deny   ip 10.10.90.0 0.0.0.255 192.168.0.0 0.0.255.255
 remark allow guest DHCP/DNS to DC1
 permit udp 10.10.90.0 0.0.0.255 host 10.10.20.10 eq 53
 permit udp 10.10.90.0 0.0.0.255 host 10.10.20.10 eq 67
 permit ip 10.10.90.0 0.0.0.255 any
!
! ---- VTY mgmt: only TECH VLAN can SSH to switches --------
ip access-list standard VTY-MGMT
 permit 10.10.50.0 0.0.0.255
 deny   any
