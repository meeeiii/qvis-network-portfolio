enable
configure terminal
!
hostname R1
no ip domain-lookup
ip domain-name qvis.local
!
! ---- Local admin account (replaces deprecated enable password) ----
username netadmin privilege 15 secret 9 $9$REPLACE-WITH-STRONG-SECRET
enable secret 9 $9$REPLACE-WITH-STRONG-SECRET
service password-encryption
!
! ---- Banners ----
banner motd ^
================================================================
  QVIS Technology Solutions - Authorised Access Only
  Disconnect immediately if you are not an authorised user.
  All access is monitored and logged.
================================================================
^
!
! ---- SSH hardening ----
crypto key generate rsa modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
!
line con 0
 logging synchronous
 login local
 exec-timeout 10 0
line vty 0 4
 transport input ssh
 login local
 exec-timeout 10 0
 access-class VTY-MGMT in
line vty 5 15
 transport input none
!
! ============================================================
! INTERFACES
! ============================================================
!
! ---- OUTSIDE: ISP handoff ----
interface GigabitEthernet0/0
 description >> ISP HANDOFF - 203.0.113.0/30 <<
 ip address 203.0.113.2 255.255.255.252
 ip nat outside
 ip access-group INET-IN in
 no shutdown
!
! ---- INSIDE: routed P2P /30 to CORE-SW1 ----
interface GigabitEthernet0/1
 description >> P2P to CORE-SW1 Gi1/0/24 <<
 ip address 10.10.0.1 255.255.255.252
 ip nat inside
 no shutdown
!
interface GigabitEthernet0/2
 description >> spare <<
 shutdown
!
! ============================================================
! ROUTING
! ============================================================
!
! ---- Default route to ISP ----
ip route 0.0.0.0 0.0.0.0 203.0.113.1
!
! ---- Static routes for every internal VLAN, next-hop CORE-SW1 ----
ip route 10.10.10.0 255.255.255.0 10.10.0.2
ip route 10.10.20.0 255.255.255.0 10.10.0.2
ip route 10.10.30.0 255.255.255.0 10.10.0.2
ip route 10.10.40.0 255.255.255.0 10.10.0.2
ip route 10.10.50.0 255.255.255.0 10.10.0.2
ip route 10.10.60.0 255.255.255.0 10.10.0.2
ip route 10.10.70.0 255.255.255.0 10.10.0.2
ip route 10.10.80.0 255.255.255.0 10.10.0.2
ip route 10.10.90.0 255.255.255.0 10.10.0.2
!
! Could be replaced with single summary: ip route 10.10.0.0 255.255.0.0 10.10.0.2
! Kept granular here for portfolio readability and easier blackholing.
!
! ============================================================
! NAT  (PAT / overload on Gi0/0)
! ============================================================
!
ip access-list standard NAT-INSIDE
 permit 10.10.10.0 0.0.0.255
 permit 10.10.20.0 0.0.0.255
 permit 10.10.30.0 0.0.0.255
 permit 10.10.40.0 0.0.0.255
 permit 10.10.50.0 0.0.0.255
 permit 10.10.60.0 0.0.0.255
 permit 10.10.70.0 0.0.0.255
 permit 10.10.80.0 0.0.0.255
 permit 10.10.90.0 0.0.0.255
 deny   any
!
ip nat inside source list NAT-INSIDE interface GigabitEthernet0/0 overload
!
! ============================================================
! ACLs
! ============================================================
!
! ---- INET-IN: applied inbound on GI0/0 (from Internet) ----
! Block obviously bad traffic from the internet; rely on stateful return for users.
ip access-list extended INET-IN
 remark === drop spoofed traffic claiming our own internal source ===
 deny   ip 10.0.0.0 0.255.255.255 any
 deny   ip 172.16.0.0 0.15.255.255 any
 deny   ip 192.168.0.0 0.0.255.255 any
 remark === drop bogons we should never see inbound ===
 deny   ip 127.0.0.0 0.255.255.255 any
 deny   ip 169.254.0.0 0.0.255.255 any
 deny   ip 224.0.0.0 15.255.255.255 any
 remark === allow established sessions back to inside ===
 permit tcp any any established
 permit udp any any
 permit icmp any any echo-reply
 permit icmp any any unreachable
 permit icmp any any time-exceeded
 remark === allow ISP ping for tshoot ===
 permit icmp host 203.0.113.1 host 203.0.113.2 echo
 deny   ip any any
!
! ---- VTY-MGMT: who can SSH to the router ----
ip access-list standard VTY-MGMT
 remark only the TECH VLAN may SSH to R1
 permit 10.10.50.0 0.0.0.255
 deny   any
!
